cmake_minimum_required(VERSION 3.30)
project(resiliencia_rs VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(resiliencia_core STATIC src/model.cpp)
target_include_directories(resiliencia_core PUBLIC include)
add_executable(resiliencia-rs src/main.cpp)
target_link_libraries(resiliencia-rs PRIVATE resiliencia_core)
enable_testing()
add_executable(resiliencia-test tests/test_model.cpp)
target_link_libraries(resiliencia-test PRIVATE resiliencia_core)
add_test(NAME nucleo_modelo COMMAND resiliencia-test)
